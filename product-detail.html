function toggleSizeMode(mode) { const standardBtn = document.getElementById('btn-mode-standard'); const customBtn = document.getElementById('btn-mode-custom'); const standardBox = document.getElementById('standard-size-box'); const customBox = document.getElementById('custom-size-box'); if (mode === 'standard') { standardBtn.className = "px-3 py-1 text-[11px] md:text-xs font-bold rounded-md bg-white text-primary shadow-sm transition-all border border-slate-200"; customBtn.className = "px-3 py-1 text-[11px] md:text-xs font-bold rounded-md text-slate-500 hover:text-slate-700 transition-all border border-transparent"; standardBox.classList.remove('hidden'); customBox.classList.add('hidden'); window.currentSizeMode = 'standard'; } else { customBtn.className = "px-3 py-1 text-[11px] md:text-xs font-bold rounded-md bg-white text-primary shadow-sm transition-all border border-slate-200"; standardBtn.className = "px-3 py-1 text-[11px] md:text-xs font-bold rounded-md text-slate-500 hover:text-slate-700 transition-all border border-transparent"; standardBox.classList.add('hidden'); customBox.classList.remove('hidden'); window.currentSizeMode = 'custom'; document.querySelectorAll('.size-btn').forEach(b => { b.classList.remove('border-2', 'border-primary', 'bg-primary', 'text-white', 'shadow-md', 'shadow-[0_4px_12px_rgba(37,99,235,0.2)]'); b.classList.add('border-slate-300', 'bg-white'); const mainText = b.querySelector('.size-main-text'); if(mainText) mainText.className = "size-main-text font-black text-sm mb-1 transition-colors text-slate-600 group-hover:text-primary"; const subText = b.querySelector('.size-sub-text'); if(subText) subText.className = "size-sub-text text-xs tracking-wide mt-0.5 transition-colors text-slate-500 group-hover:text-blue-600"; }); } } window.currentSizeMode = 'standard';
